cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(OfficeConverterTests LANGUAGES CXX)
endif()
enable_testing()
find_package(libzip CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(office_converter_tests office_converter_test.cpp)
target_compile_features(office_converter_tests PRIVATE cxx_std_20)
target_link_libraries(office_converter_tests PRIVATE libzip::zip Threads::Threads)
if(MSVC)
    target_compile_options(office_converter_tests PRIVATE /utf-8 /W4)
else()
    target_compile_options(office_converter_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME office_converter COMMAND office_converter_tests)
set_tests_properties(office_converter PROPERTIES TIMEOUT 30)
